def build_csr_vectorized(aP, aL, aR, aB, aT):
    ny, nx = aP.shape
    
    # Flatten the arrays
    d_P = aP.flatten()
    d_L = -aL.flatten()[1:]    # Offset -1
    d_R = -aR.flatten()[:-1]   # Offset +1
    d_B = -aB.flatten()[nx:]   # Offset -nx
    d_T = -aT.flatten()[:-nx]  # Offset +nx
    
    # Build sparse matrix instantly using diagonals
    A = sp.diags([d_B, d_L, d_P, d_R, d_T], 
                 offsets=[-nx, -1, 0, 1, nx], 
                 format='csr')
    return A

def build_csr_from_5pt(aP, aL, aR, aB, aT):
    ny, nx = aP.shape
    N = ny * nx
    rows, cols, data = [], [], []

    def idx(j, i):
        return j * nx + i

    for j in range(ny):
        for i in range(nx):
            p = idx(j, i)

            # diagonal
            rows.append(p); cols.append(p); data.append(aP[j, i])

            # neighbors (matrix uses -a_nb in your formulation)
            if i > 0:
                rows.append(p); cols.append(idx(j, i-1)); data.append(-aL[j, i])
            if i < nx - 1:
                rows.append(p); cols.append(idx(j, i+1)); data.append(-aR[j, i])
            if j > 0:
                rows.append(p); cols.append(idx(j-1, i)); data.append(-aB[j, i])
            if j < ny - 1:
                rows.append(p); cols.append(idx(j+1, i)); data.append(-aT[j, i])

    A = sp.csr_matrix((data, (rows, cols)), shape=(N, N))
    return A

def solve_bicgstab_ilu(aP, aL, aR, aB, aT, S, phi0=None,
                       rtol=1e-8, atol=1e-12, maxiter=500):
    ny, nx = aP.shape
    # A = build_csr_from_5pt(aP, aL, aR, aB, aT)
    A = build_csr_vectorized(aP, aL, aR, aB, aT)
    b = S.reshape(-1)

    # ILU preconditioner (built from CSC)
    A_csc = A.tocsc()
    ilu = spla.spilu(A_csc, drop_tol=1e-4, fill_factor=10)
    M = spla.LinearOperator(A.shape, matvec=ilu.solve)

    x0 = None if phi0 is None else phi0.reshape(-1)

    # x, info = spla.bicgstab(A, b, x0=x0, M=M, rtol=rtol, atol=atol, maxiter=maxiter)
    x, info = spla.bicgstab(A, b, x0=x0, rtol=rtol, atol=atol, maxiter=maxiter)

    if info != 0:
        # info>0: no convergence in maxiter, info<0: breakdown
        raise RuntimeError(f"BiCGStab failed, info={info}")

    return x.reshape(ny, nx)

def solve_pressure_amg(aP_p, aL_p, aR_p, aB_p, aT_p, b_p, p0=None,
                       rtol=1e-10, maxiter=300):
    # A = build_csr_from_5pt(aP_p, aL_p, aR_p, aB_p, aT_p)
    A = build_csr_vectorized(aP_p, aL_p, aR_p, aB_p, aT_p)
    b = b_p.reshape(-1)

    ml = pyamg.smoothed_aggregation_solver(A)     # AMG hierarchy
    M = ml.aspreconditioner(cycle='V')            # preconditioner for Krylov

    x0 = None if p0 is None else p0.reshape(-1)

    # If matrix is SPD use CG; otherwise use BiCGStab
    x, info = spla.cg(A, b, x0=x0, M=M, rtol=rtol, atol=1e-14, maxiter=maxiter)
    # x, info = spla.bicgstab(A, b, x0=x0, M=M, rtol=rtol, atol=1e-14, maxiter=maxiter)

    if info != 0:
        raise RuntimeError(f"Pressure solve failed, info={info}")

    return x.reshape(aP_p.shape)
